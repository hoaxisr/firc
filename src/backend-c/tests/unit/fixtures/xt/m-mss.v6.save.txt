:PREROUTING ACCEPT [0:0]
:INPUT ACCEPT [0:0]
:FORWARD ACCEPT [0:0]
:OUTPUT ACCEPT [0:0]
:POSTROUTING ACCEPT [0:0]

:_NDM_HOTSPOT_PRERT - [0:0]
:_NDM_QOS - [0:0]

-A _NDM_HOTSPOT_PRERT -i br0 -m mark --mark 0x0/0xfffffff -j MARK --set-xmark 0xffffaaa/0xffffffff
-A _NDM_QOS -o ppp0 -j RETURN
